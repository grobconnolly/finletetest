    toggle.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
    if (open && !was) links.querySelector('a').focus({ preventScroll: true });
    else if (!open && was && restoreFocus) toggle.focus({ preventScroll: true });
  }
  toggle.addEventListener('click', function () { setOpen(!nav.classList.contains('is-open'), true); });
  links.querySelectorAll('a').forEach(function (a) { a.addEventListener('click', function () { setOpen(false); }); });
  nav.querySelectorAll('.nav-actions a').forEach(function (a) { a.addEventListener('click', function () { setOpen(false); }); });
  document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && nav.classList.contains('is-open')) setOpen(false, true); });
  document.addEventListener('pointerdown', function (e) {
    if (!nav.classList.contains('is-open') || nav.contains(e.target)) return;
    setOpen(false);
    // the tap that dismisses the menu must not also activate the link under it
    var swallow = function (ev) { ev.preventDefault(); ev.stopPropagation(); };
    document.addEventListener('click', swallow, { capture: true, once: true });
    setTimeout(function () { document.removeEventListener('click', swallow, true); }, 700);
  });
  nav.addEventListener('focusout', function (e) { if (nav.classList.contains('is-open') && e.relatedTarget && !nav.contains(e.relatedTarget)) setOpen(false); });
  if (window.matchMedia) {
    var mq = window.matchMedia('(min-width: 961px)');
    var onMq = function () { setOpen(false); };
    if (mq.addEventListener) mq.addEventListener('change', onMq); else if (mq.addListener) mq.addListener(onMq);
  }

  // the funded list is a tab stop only while it scrolls sideways (phones)
  var minis = document.querySelector('.minis');
  var syncMinis = function () { if (minis.scrollWidth > minis.clientWidth) minis.setAttribute('tabindex', '0'); else minis.removeAttribute('tabindex'); };
  syncMinis();
  window.addEventListener('resize', syncMinis);

  // a focused offering card shows whole, ring included, wherever it fits under the nav
  document.querySelectorAll('.card-cta').forEach(function (a) {
    a.addEventListener('focus', function () {
      var card = a.closest('.card');
      if (a.matches(':focus-visible') && card.offsetHeight + 97 <= window.innerHeight) card.scrollIntoView({ block: 'nearest' });
    });
  });

  // ----- analytics hook: every call to action pushes one dataLayer event -----
  document.addEventListener('click', function (e) {
    var el = e.target.closest ? e.target.closest('[data-cta]') : null;
    if (el) (window.dataLayer = window.dataLayer || []).push({ event: 'cta_click', cta: el.getAttribute('data-cta') });
  });
})();
</script>
