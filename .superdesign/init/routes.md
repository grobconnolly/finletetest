## `nextjs/next.config.ts`
```ts
import type { NextConfig } from "next";

/**
 * Pixel-perfect static mirror of finlete.com.
 *
 * Every page is a captured Framer HTML file at its exact URL path inside
 * `public/`:
 *   /                          → public/index.html
 *   /about                     → public/about/index.html
 *   /athletes                  → public/athletes/index.html
 *   /avila, /escobar, /ornelas,
 *   /padilla, /santos, /teodo  → public/<slug>/index.html
 *   /athlete-landingpage-2026  → public/athlete-landingpage-2026/index.html
 *   /legal/terms-of-service    → public/legal/terms-of-service/index.html
 *   /legal/privacy-policy      → public/legal/privacy-policy/index.html
 *
 * Assets (images, fonts) are served from public/images/ and public/fonts/,
 * referenced by absolute paths from inside each mirrored HTML file.
 *
 * `vercel.json` sets framework=nextjs so Vercel runs `next build` (rather
 * than treating this as a static-HTML-only project).
 */
const nextConfig: NextConfig = {
  async rewrites() {
    return [
      // Static redesign prototype lives in public/redesign; Next.js only
      // serves public files at their exact path, so map the clean URL.
      { source: "/redesign", destination: "/redesign/index.html" },
    ];
  },
};

export default nextConfig;
```

## `scripts/sync-redesign.py`
```python
#!/usr/bin/env python3
"""Copy redesign/ (the source of truth) into nextjs/public/redesign.

redesign/ is self-contained and uses document-relative paths (img/, fonts/,
styles.css), so it opens from the file system and from any folder URL.
Next.js serves the page at /redesign with no trailing slash, where relative
paths would resolve against the site root, so the deployed copy gets
absolute /redesign/ paths.

Run from anywhere:  python3 scripts/sync-redesign.py
"""
import re
import shutil
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
SRC = ROOT / "redesign"
DST = ROOT / "nextjs" / "public" / "redesign"

if DST.exists():
    shutil.rmtree(DST)
shutil.copytree(SRC, DST)

page = DST / "index.html"
html = page.read_text()
html = re.sub(r'(["\s,])(img|fonts)/', r"\1/redesign/\2/", html)
html = html.replace('href="styles.css"', 'href="/redesign/styles.css"')
assert 'src="img/' not in html and 'href="fonts/' not in html
page.write_text(html)
print(f"synced {sum(1 for _ in DST.rglob('*') if _.is_file())} files to {DST.relative_to(ROOT)}")
```
